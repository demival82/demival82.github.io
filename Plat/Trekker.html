/** вставить в конец JavaScript * Russian Text HTML Normalizer (RTHN) * Версия 1.1.0 * * Библиотека для корректного отображения русского текста в HTML: * - Исправление проблем с кодировкой * - Нормализация кавычек и пунктуации * - Автоматическая расстановка неразрывных пробелов * - Коррекция HTML-сущностей * - Исправление типичных опечаток */ class RussianHTMLNormalizer { constructor(options = {}) { this.defaultOptions = { fixEncoding: true, // исправлять проблемы кодировки normalizeQuotes: true, // нормализовать кавычки useNbsp: true, // использовать неразрывные пробелы fixCommonTypos: true, // исправлять частые опечатки fixHtmlEntities: true, // исправлять HTML-сущности processWholePage: false // обрабатывать всю страницу автоматически }; this.options = { ...this.defaultOptions, ...options }; // Паттерны для неразрывных пробелов this.nbspPatterns = [ /(\s)([а-яёА-ЯЁ]{1,2})\s+/gi, // короткие слова /(\s[а-яёА-ЯЁ]+)\s([а-яёА-ЯЁ]{1,2})\s/gi, // перед короткими словами /([«])\s+/g, // после открывающих кавычек /\s+([»])/g, // перед закрывающими кавычек /(\d+)\s+(года?|лет|месяц|день)/gi, // числа с единицами измерения /(^|\s)([A-Za-zА-Яа-яёЁ]+)\s-\s/g // дефисы между словами ]; // Таблица замены HTML-сущностей this.htmlEntitiesMap = { '"': '"', '&APOS;': "'", '"': '"', ''': "'", '‘': '‘', '’': '’', '“': '“', '”': '”' }; // Распространенные опечатки this.commonTypos = { 'зделать': 'сделать', 'здесь': 'сдесь', 'вообщем': 'в общем', 'ихний': 'их', 'еёный': 'её', 'скучать по вас': 'скучать по вам' }; // Инициализация, если нужно обработать всю страницу if (this.options.processWholePage && typeof document !== 'undefined') { this.processDocument(); } } /** * Основная функция нормализации текста * @param {string} text - Входной текст * @returns {string} Нормализованный текст */ normalize(text) { if (typeof text !== 'string') return text; let normalized = text; // Исправление кодировки if (this.options.fixEncoding) { normalized = this.fixEncoding(normalized); } // Нормализация кавычек if (this.options.normalizeQuotes) { normalized = this.normalizeQuotes(normalized); } // Исправление опечаток if (this.options.fixCommonTypos) { normalized = this.fixCommonTypos(normalized); } // Исправление HTML-сущностей if (this.options.fixHtmlEntities) { normalized = this.fixHtmlEntities(normalized); } // Добавление неразрывных пробелов if (this.options.useNbsp) { normalized = this.addNbsp(normalized); } return normalized; } /** * Исправление проблем с кодировкой * @param {string} text * @returns {string} */ fixEncoding(text) { // Заменяем распространенные проблемы с кодировкой return text .replace(/ÃÂ°/g, 'ё') .replace(/Ã/g, '') .replace(/Â°/g, '') .replace(/Ã/g, '×') .replace(/Ã©/g, 'é') .replace(/â/g, '–') .replace(/â/g, '—') .replace(/â/g, '’') .replace(/â/g, '‘') .replace(/â/g, '”') .replace(/â/g, '“'); } /** * Нормализация кавычек * @param {string} text * @returns {string} */ normalizeQuotes(text) { // Заменяем прямые кавычки на «ёлочки» return text .replace(/"([^"]+)"/g, '«$1»') .replace(/'([^']+)'/g, '«$1»') .replace(/(\s|^)"(\S)/g, '$1«$2') // Открывающие кавычки .replace(/(\S)"(\s|$|[.,!?])/g, '$1»$2'); // Закрывающие кавычки } /** * Добавление неразрывных пробелов * @param {string} text * @returns {string} */ addNbsp(text) { let result = text; this.nbspPatterns.forEach(pattern => { result = result.replace(pattern, (match, p1, p2) => { if (p2) { return `${p1}${p2}\u00A0`; } return `${p1}\u00A0`; }); }); return result; } /** * Исправление HTML-сущностей * @param {string} text * @returns {string} */ fixHtmlEntities(text) { let result = text; for (const [entity, replacement] of Object.entries(this.htmlEntitiesMap)) { result = result.replace(new RegExp(entity, 'g'), replacement); } return result; } /** * Исправление распространенных опечаток * @param {string} text * @returns {string} */ fixCommonTypos(text) { let result = text; for (const [typo, correction] of Object.entries(this.commonTypos)) { const regex = new RegExp(`\\b${typo}\\b`, 'gi'); result = result.replace(regex, correction); } return result; } /** * Обработка всего HTML-документа */ processDocument() { if (typeof document === 'undefined') return; const walker = document.createTreeWalker( document.body, NodeFilter.SHOW_TEXT, null, false ); let node; while (node = walker.nextNode()) { if (node.nodeValue.trim()) { const normalized = this.normalize(node.nodeValue); if (normalized !== node.nodeValue) { node.nodeValue = normalized; } } } } /** * Обработка конкретного HTML-элемента * @param {HTMLElement} element */ processElement(element) { if (typeof element === 'string') { element = document.querySelector(element); } if (!element) return; const walker = document.createTreeWalker( element, NodeFilter.SHOW_TEXT, null, false ); let node; while (node = walker.nextNode()) { if (node.nodeValue.trim()) { const normalized = this.normalize(node.nodeValue); if (normalized !== node.nodeValue) { node.nodeValue = normalized; } } } } } // Экспорт для разных сред if (typeof module !== 'undefined' && module.exports) { module.exports = RussianHTMLNormalizer; } else if (typeof window !== 'undefined') { window.RussianHTMLNormalizer = RussianHTMLNormalizer; // Автоматическая инициализация при загрузке страницы document.addEventListener('DOMContentLoaded', () => { if (window.RTHNAutoInit !== false) { const normalizer = new RussianHTMLNormalizer({ processWholePage: true }); } }); }
